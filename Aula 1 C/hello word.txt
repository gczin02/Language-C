#include <stdio.h>

int main() {
    float potencia, horas_dia, consumo;

    printf("Potencia do equipamento (W): ");
    scanf("%f", &potencia);

    printf("Horas de uso por dia: ");
    scanf("%f", &horas_dia);

    consumo = (potencia * horas_dia * 30) / 1000;

    printf("Consumo mensal: %.2f kWh\n", consumo);

    return 0;
}





#include <stdio.h>

int main() {
    float peso, altura, imc;

    printf("Digite seu peso em kg: ");
    scanf("%f", &peso);

    printf("Digite sua altura em metros: ");
    scanf("%f", &altura);

    imc = peso / (altura * altura);

    printf("\nSeu IMC e: %.2f\n", imc);

    if (imc < 18.5) {
        printf("Classificacao: Abaixo do peso\n");
    } else if (imc < 25.0) {
        printf("Classificacao: Peso normal\n");
    } else if (imc < 30.0) {
        printf("Classificacao: Sobrepeso\n");
    } else {
        printf("Classificacao: Obesidade\n");
    }

    return 0;
}







#include <stdio.h>
#include <math.h>

int main() {
    float largura, comprimento, valorCaixa;
    float area;
    int quantidadeCaixas;
    float custoTotal;

    printf("Qual e a largura da area (em metros)? ");
    scanf("%f", &largura);

    printf("Qual e o comprimento da area (em metros)? ");
    scanf("%f", &comprimento);

    printf("Qual e o valor de cada caixa? ");
    scanf("%f", &valorCaixa);

    area = largura * comprimento;

    quantidadeCaixas = (int)ceil(area / 2.5);

    custoTotal = quantidadeCaixas * valorCaixa;

    printf("\nArea total a ser revestida: %.2f m2\n", area);
    printf("Quantidade de caixas necessarias: %d\n", quantidadeCaixas);
    printf("Custo total da compra: R$ %.2f\n", custoTotal);

    return 0;
}
